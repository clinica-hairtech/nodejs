5) Helper mv-seguro com verificação:
```bash
mover_seguro() {
  local src="$1"
  local dst_dir="$2"
  [ ! -f "$src" ] && return 1
  [ ! -d "$dst_dir" ] && mkdir -p "$dst_dir"
  local base=$(basename "$src")
  local dst="$dst_dir/$base"
  # Se ja existe destino, gera nome com timestamp
  [ -e "$dst" ] && dst="$dst_dir/$(date +%Y%m%d)_$base"
  local h1=$(shasum -a 256 "$src" | awk '{print $1}')
  mv "$src" "$dst" 2>/dev/null || { echo "FAIL_MV: $src -> $dst" >> ~/HairTech_org_log.txt; return 2; }
  local h2=$(shasum -a 256 "$dst" | awk '{print $1}')
  if [ "$h1" = "$h2" ]; then
    echo "OK: $src -> $dst" >> ~/HairTech_org_log.txt
    return 0
  else
    mv "$dst" "$src" 2>/dev/null
    echo "FAIL_HASH: $src -> $dst (hash divergiu)" >> ~/HairTech_org_log.txt
    return 3
  fi
}
export -f mover_seguro
```

import http.server
import json
import subprocess
import hashlib
import os

SECRET = os.environ.get("HAIRTECH_WEBHOOK_SECRET", "")
ALLOWED_COMMANDS = ["ls", "find", "shasum", "du", "df", "mv", "mkdir", "echo", "cat", "head", "tail", "wc"]

class Handler(http.server.BaseHTTPRequestHandler):
    def do_POST(self):
        if self.path != "/exec":
            self.send_response(404); self.end_headers(); return
        length = int(self.headers.get("Content-Length", 0))
        data = json.loads(self.rfile.read(length))
        if data.get("secret") != SECRET:
            self.send_response(401); self.end_headers(); return
        cmd = data.get("cmd", "")
        # Whitelist do primeiro comando
        first = cmd.strip().split()[0] if cmd.strip() else ""
        if first not in ALLOWED_COMMANDS:
            self.send_response(403)
            self.end_headers()
            self.wfile.write(json.dumps({"error": f"cmd nao permitido: {first}"}).encode())
            return
        try:
            r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=60)
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.end_headers()
            self.wfile.write(json.dumps({"stdout": r.stdout[:50000], "stderr": r.stderr[:5000], "code": r.returncode}).encode())
        except Exception as e:
            self.send_response(500); self.end_headers()
            self.wfile.write(json.dumps({"error": str(e)}).encode())

server = http.server.HTTPServer(("127.0.0.1", 9876), Handler)
print("HairTech webhook ativo em :9876")
server.serve_forever()
PYEOF
