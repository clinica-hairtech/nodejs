PASSO 4 - Subir OpenClaw container:

```bash
docker stop openclaw-mac 2>/dev/null
docker rm openclaw-mac 2>/dev/null

docker run -d \
  --name openclaw-mac \
  --restart unless-stopped \
  -p 127.0.0.1:18790:18790 \
  -v ~/.openclaw:/home/node/.openclaw \
  --add-host=host.docker.internal:host-gateway \
  openclaw/openclaw:2026.5.7 \
  --config /home/node/.openclaw/config/openclaw.json

sleep 10
docker logs openclaw-mac | tail -30
docker ps | grep openclaw-mac
```

PASSO 6 - Conectar com OpenClaw da VPS (irmao):

```bash
# Adiciona Mac OpenClaw como peer do VPS
# (peering permite Dr. dar comando em qualquer um, e eles se coordenam)
cat > ~/.openclaw/config/peers.json <<'PEEREOF'
{
  "peers": [
    {
      "name": "hairtech-openclaw-vps",
      "url": "https://claw.hairtech.org",
      "role": "primary",
      "scopes": ["whatsapp", "pacientes", "agenda", "financeiro"]
    }
  ],
  "self_role": "secondary",
  "self_scopes": ["mac-files", "icloud", "calendar-pessoal", "fotos"]
}
PEEREOF

docker restart openclaw-mac
sleep 5
```
